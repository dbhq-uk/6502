using System.Reflection;
using Dbhq.Bench;
using Dbhq.Cpu6502.TestSupport;
using Dbhq.Machines.Nes;

// Task 3b scratch: plays Lan Master by bench/nes-speed/lan-master-play.json from power on to the script's
// last frame of play in each region, and counts the PPU's lines by phase, path, PPUMASK and sprites.
byte[] rom = RepoPaths.ReadChecked(Pins.NesHomebrewPath, Pins.NesHomebrewSha256);
string scriptPath = Path.Combine(RepoPaths.Root, "bench", "nes-speed", "lan-master-play.json");
bool oracle = args.Contains("--oracle");
Probe.TierA = args.Contains("--tier-a");
foreach (Region region in new[] { Region.Ntsc, Region.Pal })
{
    var script = PlayScript.Load(scriptPath, region.Name);
    var names = script.PhaseNames;
    int PhaseIndex(long frame) => names.ToList().IndexOf(script.PhaseAt(frame));
    Probe.Reset();
    var nes = new Nes(Cartridge.Load(rom), region, new NesOptions { PerDotReference = oracle });
    var ppu = nes.Bus.Ppu;
    nes.PowerOn();
    var frames = new long[names.Count];
    while (ppu.Frame < script.Limit)
    {
        int p = PhaseIndex(ppu.Frame);
        Probe.Phase = p;
        frames[p]++;
        nes.SetButtons(0, script.MaskAt(ppu.Frame));
        nes.RunFrames(1);
    }

    long fastTotal = (long)typeof(Ppu).GetProperty("FastLines", BindingFlags.NonPublic | BindingFlags.Instance)!.GetValue(ppu)!;
    long exactTotal = (long)typeof(Ppu).GetProperty("ExactLines", BindingFlags.NonPublic | BindingFlags.Instance)!.GetValue(ppu)!;
    long probeFast = Probe.Lines.Sum(l => l[0] + l[2]);
    long probeExact = Probe.Lines.Sum(l => l[1] + l[3] + l[4]);
    Console.WriteLine($"## {region.Name}{(oracle ? " (per-dot reference)" : "")}{(Probe.TierA ? " (scratch: also sprites-on lines with no sprite on them)" : "")}: from power on to frame {ppu.Frame} (limit {script.Limit}); Ppu.FastLines {fastTotal} ExactLines {exactTotal}; probe fast {probeFast} exact {probeExact} {(fastTotal == probeFast && exactTotal == probeExact ? "agree" : "DISAGREE")}");
    Console.WriteLine("phase | frames | visible lines | fast % of visible | idle lines (VBlank, post-render) fast % | all lines fast % | mean sprites on a visible line (OAM) | visible lines with a sprite % | with 0/1/2/3-8/9+ sprites % | PPUMASK of visible lines: off / bg / sprites / both %");
    for (int p = 0; p < names.Count; p++)
    {
        // Phases with the same name (none now) would be summed by the caller.
        long[] c = Probe.Lines[p];
        long visible = c[0] + c[1];
        long idle = c[2] + c[3];
        long all = visible + idle + c[4];
        long[] s = Probe.Sprites[p];
        long sv = s.Sum();
        double mean = sv == 0 ? 0 : Enumerable.Range(0, 10).Sum(n => (double)s[n] * Math.Min(n, 9)) / sv;
        long[] m = Enumerable.Range(0, 4).Select(i => Probe.ByMask[p][i][0] + Probe.ByMask[p][i][1]).ToArray();
        string Pct(long a, long b) => b == 0 ? "-" : $"{100.0 * a / b:F1}";
        Console.WriteLine($"{names[p]} | {frames[p]} | {visible} | {Pct(c[0], visible)} | {idle} {Pct(c[2], idle)} | {Pct(c[0] + c[2], all)} | {mean:F2} | {Pct(sv - s[0], sv)} | {Pct(s[0], sv)}/{Pct(s[1], sv)}/{Pct(s[2], sv)}/{Pct(s[3] + s[4] + s[5] + s[6] + s[7] + s[8], sv)}/{Pct(s[9], sv)} | {Pct(m[0], visible)} / {Pct(m[1], visible)} / {Pct(m[2], visible)} / {Pct(m[3], visible)}");
    }
}

namespace Dbhq.Machines.Nes;

// Task 3b scratch probe, never committed to src: counts each line the PPU finishes by the path that
// ran it, the phase of the play session (set by the harness), PPUMASK, and the sprites of OAM on it.
public static class Probe
{
    public static int Phase;
    public static bool TierA;
    public const int Phases = 8;
    // [phase][0 visible fast, 1 visible exact, 2 idle fast, 3 idle exact (VBlank and post-render), 4 pre-render]
    public static readonly long[][] Lines = Enumerable.Range(0, Phases).Select(_ => new long[5]).ToArray();
    // visible lines by PPUMASK layers (bits 3 and 4 of the mask: 0, 8, 16, 24) and path: [phase][layers/8][fast 0 or exact 1]
    public static readonly long[][][] ByMask = Enumerable.Range(0, Phases).Select(_ => Enumerable.Range(0, 4).Select(_ => new long[2]).ToArray()).ToArray();
    // visible lines by the sprites of OAM on the line (0 to 8, then 9 for more), whatever the mask: [phase][n]
    public static readonly long[][] Sprites = Enumerable.Range(0, Phases).Select(_ => new long[10]).ToArray();
    // the same, only the visible lines the sprite layer is on for
    public static readonly long[][] SpritesOn = Enumerable.Range(0, Phases).Select(_ => new long[10]).ToArray();
    // visible lines with the sprite layer on, by path and whether a sprite is on the line: [phase][0 none or 1 some][fast 0 or exact 1]
    public static void Reset() { foreach (var l in Lines) Array.Clear(l); foreach (var p in ByMask) foreach (var l in p) Array.Clear(l); foreach (var l in Sprites) Array.Clear(l); foreach (var l in SpritesOn) Array.Clear(l); }
}

public sealed partial class Ppu
{
    internal void ProbeLine(int line, bool fast)
    {
        long[] c = Probe.Lines[Probe.Phase];
        if (line >= FrameBuffer.Height)
        {
            if (line == _preRenderLine) c[4]++;
            else c[fast ? 2 : 3]++;
            return;
        }

        c[fast ? 0 : 1]++;
        int layers = (_mask & 0x18) >> 3;
        Probe.ByMask[Probe.Phase][layers][fast ? 0 : 1]++;
        int height = (_ctrl & 0x20) != 0 ? 16 : 8;
        int n = 0;
        for (int i = 0; i < 64; i++)
        {
            int y = _oam[i * 4];
            if (y < 0xF0 && line >= y + 1 && line < y + 1 + height) n++;
        }

        Probe.Sprites[Probe.Phase][Math.Min(n, 9)]++;
        if ((_mask & 0x10) != 0) Probe.SpritesOn[Probe.Phase][Math.Min(n, 9)]++;
    }
}
